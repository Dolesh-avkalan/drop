def _validate_filename(name: object) -> str:
    """Validate a fixture filename, returning it if valid.

    A valid filename is a non-empty string that is not ``"."``, ``".."``,
    or ``"root"`` and that contains no slash, backslash, or NUL.
    """
    if not isinstance(name, str):
        raise ValueError(f"filename must be a string, got {type(name).__name__}")
    if name == "":
        raise ValueError("filename must be non-empty")
    if name in _INVALID_FILENAME_TOKENS:
        raise ValueError(f"filename {name!r} is reserved")
    for char in _INVALID_FILENAME_CHARS:
        if char in name:
            raise ValueError(f"filename {name!r} must not contain {char!r}")
    return name

def _validate_max_entries(max_entries: object) -> None:
    """Validate an optional ``max_entries``: ``None`` or ``int >= 1``."""
    if max_entries is None:
        return
    if isinstance(max_entries, bool) or not isinstance(max_entries, int):
        raise ValueError(
            f"max_entries must be a positive int or None, "
            f"got {type(max_entries).__name__}"
        )
    if max_entries < 1:
        raise ValueError(f"max_entries must be >= 1, got {max_entries}")


def _validate_scan_max_entries(max_entries: object) -> None:
    """Validate a mandatory ``scan_children`` budget.

    ``max_entries`` is a required positive ``int`` in ``1..MAX_SCAN_ENTRIES``;
    ``bool`` (including ``True``/``False``) and ``None`` are rejected.

class InMemoryAdapter:
    """A read-only adapter over flat synthetic fixtures held in memory.

    ``files`` maps a filename to its exact bytes. The filename is also the
    file's ``file_id``. A synthetic root directory (id ``"root"``) is the
    single parent of every fixture file.
    """

    def __init__(self, source_id: str, files: Mapping[str, bytes]) -> None:
        if not isinstance(source_id, str) or source_id == "":
            raise ValueError("source_id must be a non-empty string")
        if not isinstance(files, Mapping):
            raise ValueError(
                f"files must be a mapping, got {type(files).__name__}"
            )

        # Validate first, then copy, so a bad mapping never half-populates us.
        validated: dict[str, bytes] = {}
        for name, payload in files.items():
            _validate_filename(name)
            if not isinstance(payload, bytes):
                raise ValueError(
                    f"payload for {name!r} must be bytes, "
                    f"got {type(payload).__name__}"
                )
            validated[name] = payload

        self._source_id = source_id
        # Copy the mapping so later external mutation cannot affect us.
        self._files: dict[str, bytes] = dict(validated)

    @property
    def source_id(self) -> str:
        return self._source_id

    # -- internals ---------------------------------------------------------

    def _root_ref(self) -> FileRef:
        return FileRef(self._source_id, ROOT_ID)

    def _root_metadata(self) -> FileMetadata:
        return FileMetadata(
            ref=self._root_ref(),
            name="",
            parent_id=None,
            kind="directory",
            size_bytes=None,
        )

    def _file_metadata(self, name: str) -> FileMetadata:
        return FileMetadata(
            ref=FileRef(self._source_id, name),
            name=name,
            parent_id=ROOT_ID,
            kind="file",
            size_bytes=len(self._files[name]),
        )

    def _check_source(self, ref: FileRef) -> None:
        """Reject a mismatched source_id. Always runs before file_id checks."""
        if not isinstance(ref, FileRef):
            raise TypeError(f"expected FileRef, got {type(ref).__name__}")
        if ref.source_id != self._source_id:
            raise SourceMismatchError(
                f"ref source_id {ref.source_id!r} does not match "
                f"adapter source_id {self._source_id!r}"
            )

    # -- contract ----------------------------------------------------------

    def stat(self, ref: FileRef) -> FileMetadata:
        self._check_source(ref)
        if ref.file_id == ROOT_ID:
            return self._root_metadata()
        if ref.file_id not in self._files:
            raise NotFoundError(f"unknown file_id {ref.file_id!r}")
        return self._file_metadata(ref.file_id)

    def list_children(
        self, parent: FileRef, *, max_entries: int | None = None
    ) -> list[FileMetadata]:
        self._check_source(parent)
        _validate_max_entries(max_entries)
        if parent.file_id == ROOT_ID:
            if max_entries is not None and len(self._files) > max_entries:
                raise EnumerationLimitError(
                    max_entries,
                    f"directory {parent.file_id!r} exceeds limit of "
                    f"{max_entries} entries",
                )
            return [self._file_metadata(name) for name in sorted(self._files)]
        if parent.file_id not in self._files:
            raise NotFoundError(f"unknown file_id {parent.file_id!r}")
        raise NotDirectoryError(f"{parent.file_id!r} is not a directory")

    def scan_children(self, parent: FileRef, *, max_entries: int) -> ScanBatch:
        """Return a bounded, insertion-order prefix of the fixture children.

        ``max_entries`` is a mandatory positive ``int``. The fixture mapping is
        never sorted or fully materialized: at most ``max_entries`` names are
        taken in insertion order (a bounded prefix), and only that prefix is
        sorted by ``(name, file_id)``. Because the total length is known, an
        in-memory scan reports exact completeness: ``complete=True`` when every
        fixture entry fit in the prefix, ``complete=False`` otherwise.
        """
        self._check_source(parent)
        _validate_scan_max_entries(max_entries)
        if parent.file_id == ROOT_ID:
            total = len(self._files)
            prefix_names = list(itertools.islice(self._files, max_entries))
            items = tuple(
                sorted(
                    (self._file_metadata(name) for name in prefix_names),
                    key=lambda meta: (meta.name, meta.ref.file_id),
                )
            )
            return ScanBatch(
                items=items,
                inspected=len(prefix_names),
                complete=total <= max_entries,
            )
        if parent.file_id not in self._files:
            raise NotFoundError(f"unknown file_id {parent.file_id!r}")
        raise NotDirectoryError(f"{parent.file_id!r} is not a directory")

    def read_bytes(self, ref: FileRef) -> bytes:
        self._check_source(ref)
        if ref.file_id == ROOT_ID:
            raise IsDirectoryError(f"{ROOT_ID!r} is a directory, not a file")
        if ref.file_id not in self._files:
            raise NotFoundError(f"unknown file_id {ref.file_id!r}")
        return self._files[ref.file_id]

"""Bounded read-only local filesystem storage adapter (MODULE-003).

This module implements :class:`LocalAdapter`, a read-only :class:`StorageAdapter`
backed by an operator-controlled POSIX directory tree.

Security and containment model:
- Configuration traverses from ``/`` to the target root directory component by
  component using ``O_RDONLY | O_NOFOLLOW | O_NONBLOCK`` opens anchored with
  ``dir_fd`` (deliberately WITHOUT ``O_DIRECTORY``). Each returned descriptor is
  immediately ``fstat``ed and required to be a directory (``S_ISDIR``). The
  ``O_NOFOLLOW`` alone is the atomic guard: a symlink component (final or
  intermediate) makes ``openat`` fail with ``ELOOP``, so a directory swapped for
  a symlink immediately before its open cannot redirect access outside the root.
  ``O_DIRECTORY`` is intentionally omitted because on some kernels (notably the
  gVisor sandbox used for acceptance) ``O_NOFOLLOW`` is ignored when combined
  with ``O_DIRECTORY``, which would silently follow a symlink. Because real
  no-follow semantics are required and a constant being defined is not trusted,
  the adapter probes the running kernel at construction time and fails closed
  with :class:`UnsupportedPlatformError` when ``O_NOFOLLOW`` is not actually
  enforced. The resulting root directory file descriptor is retained until
  :meth:`close`.
- All subsequent operations (`stat`, `list_children`, `read_bytes`) resolve
  canonical ``p/<relative_posix_path>`` IDs step by step using anchored
  directory descriptors (`dir_fd`) and no-follow semantics (`O_NOFOLLOW` /
  `follow_symlinks=False`). String path prefix checks are never used for
  containment.
- Only regular files and directories are supported. Symbolic links raise
  :class:`UnsafePathError` on direct access or intermediate traversal and are
  omitted from directory listings. FIFOs, sockets, and device nodes raise
  :class:`UnsupportedEntryError` without blocking (`O_NONBLOCK` on open,
  `fstatat` with `AT_SYMLINK_NOFOLLOW` on stat/list) and are omitted from
  directory listings.
- Reads enforce `max_read_bytes` via `fstat` prior to reading and during chunked
  reads by requesting at most `max_read_bytes + 1` total bytes, preventing both
  unbounded memory allocation and bypasses when a file grows concurrently after
  `fstat`.

Documented limitations and threat boundaries:
- Concurrent external writers may modify a file while it is being read; this
  adapter bounds the maximum bytes read but does not provide a coherent
  point-in-time snapshot or cryptographic checksum guarantee.
- Concurrent calls to `stat`/`list_children`/`read_bytes` racing with
  :meth:`close` on the same adapter instance are not supported; callers must
  synchronize lifecycle teardown externally if sharing an instance across
  threads.

    A defined constant is not evidence: some sandboxes (e.g. gVisor) expose

    symlink with the *exact* flags used for directory components

        base = tempfile.mkdtemp(prefix="drive_tool_nofollow_probe_")

        link = os.path.join(base, "link")
        try:

            os.symlink(target, link)
        except OSError:
            return False

        # Use the production directory-component flag helper itself, so the
        # probe can never drift from real adapter behaviour: the helper is the
        # single source of truth, resolved at call time (never cached).
        flags = _dir_open_flags()
        try:
            fd = os.open(link, flags)
        except OSError as exc:
            if exc.errno not in (errno.ELOOP, getattr(errno, "EMLINK", -1)):
                return False
        else:
            # The symlink was followed: O_NOFOLLOW is not enforced.
            os.close(fd)
            return False

        try:
            dir_fd = os.open(target, flags)
        except OSError:
            return False
        try:
            if not stat.S_ISDIR(os.fstat(dir_fd).st_mode):
                return False
        finally:
            os.close(dir_fd)
        return True
    finally:
        shutil.rmtree(base, ignore_errors=True)


def _check_platform_capabilities() -> None:
    """Verify that the runtime platform supports anchored no-follow I/O.

    Raises :class:`UnsupportedPlatformError` if any required POSIX feature is
    unavailable, or if a runtime probe shows that ``O_NOFOLLOW`` is not
    actually enforced. Weaker path-string fallbacks are never emulated.
    """
    if os.name != "posix":
        raise UnsupportedPlatformError(
            f"LocalAdapter requires a POSIX platform, got os.name={os.name!r}"
        )

    required_flags = ("O_RDONLY", "O_NOFOLLOW", "O_NONBLOCK")
    for flag_name in required_flags:
        flag_val = getattr(os, flag_name, None)
        if not isinstance(flag_val, int) or (
            flag_name != "O_RDONLY" and flag_val == 0
        ):
            raise UnsupportedPlatformError(
                f"LocalAdapter requires os.{flag_name} support on this platform"
            )

    for func_name in ("open", "close", "fstat", "stat", "scandir", "read"):
        if not hasattr(os, func_name):
            raise UnsupportedPlatformError(
                f"LocalAdapter requires os.{func_name} support on this platform"
            )

    supports_dir_fd = getattr(os, "supports_dir_fd", set())
    if os.open not in supports_dir_fd or os.stat not in supports_dir_fd:
        raise UnsupportedPlatformError(
            "LocalAdapter requires dir_fd support for os.open and os.stat"
        )

    supports_follow_symlinks = getattr(os, "supports_follow_symlinks", set())
    if os.stat not in supports_follow_symlinks:
        raise UnsupportedPlatformError(
            "LocalAdapter requires follow_symlinks=False support for os.stat"
        )

    supports_fd = getattr(os, "supports_fd", set())
    if os.scandir not in supports_fd:
        raise UnsupportedPlatformError(
            "LocalAdapter requires file descriptor support for os.scandir"
        )

    if not _no_follow_semantics_available():
        raise UnsupportedPlatformError(
            "LocalAdapter requires O_NOFOLLOW to actually reject symbolic links "
            "on this platform (verified by a runtime probe, not merely present "
            "as a constant)"
        )


def _dir_open_flags() -> int:
    """Return flags for a no-follow, non-blocking directory-component open.

    Deliberately excludes ``O_DIRECTORY``: on some kernels ``O_NOFOLLOW`` is
    ignored when combined with ``O_DIRECTORY``, which would silently follow a
    symlink. ``O_NOFOLLOW`` alone is honoured there, so it is the atomic guard;
    the caller verifies the returned descriptor's type with ``fstat``.
    """
    return (
        os.O_RDONLY
        | os.O_NOFOLLOW
        | os.O_NONBLOCK
        | getattr(os, "O_CLOEXEC", 0)
    )


def _file_open_flags() -> int:
    """Return flags for non-blocking, no-follow file opens."""
    return (
        os.O_RDONLY
        | os.O_NOFOLLOW
        | os.O_NONBLOCK
        | getattr(os, "O_CLOEXEC", 0)
    )


def _parse_file_id(file_id: object) -> tuple[str, ...]:
    """Validate and decompose ``file_id`` into canonical relative path parts.

    Returns ``()`` for the synthetic ``"root"`` ID, or a non-empty tuple of
    validated path components for ``"p/..."`` IDs. Raises
    :class:`UnsafePathError` on any malformed ID.
    """
    if not isinstance(file_id, str):
        raise UnsafePathError(
            f"file_id must be a str, got {type(file_id).__name__}"
        )
    if file_id == ROOT_ID:
        return ()
    if "\x00" in file_id:
        raise UnsafePathError(f"file_id {file_id!r} must not contain NUL")
    if "\\" in file_id:
        raise UnsafePathError(
            f"file_id {file_id!r} must not contain backslash"
        )
    if file_id.startswith("/"):
        raise UnsafePathError(
            f"file_id {file_id!r} must not be an absolute path"
        )
    if not file_id.startswith(PATH_PREFIX):
        raise UnsafePathError(
            f"file_id {file_id!r} must be {ROOT_ID!r} or start with {PATH_PREFIX!r}"
        )

    rel_path = file_id[len(PATH_PREFIX) :]
    if rel_path == "":
        raise UnsafePathError(
            f"file_id {file_id!r} must have a non-empty relative path"
        )
    if rel_path.endswith("/"):
        raise UnsafePathError(
            f"file_id {file_id!r} must not have a trailing slash"
        )

    parts = rel_path.split("/")
    for part in parts:
        if part == "":
            raise UnsafePathError(
                f"file_id {file_id!r} contains an empty path component"
            )
        if part in (".", ".."):
            raise UnsafePathError(
                f"file_id {file_id!r} contains disallowed component {part!r}"
            )

    return tuple(parts)


def _validate_root_path_string(root: str | os.PathLike[str]) -> tuple[str, list[str]]:
    """Validate the configured root path and split it into traversal parts."""
    if not isinstance(root, (str, os.PathLike)):
        raise ValueError(
            f"root must be a str or os.PathLike, got {type(root).__name__}"
        )
    try:
        raw_root = os.fspath(root)
    except TypeError as exc:
        raise ValueError(f"root must be a valid path: {exc}") from exc

    if not isinstance(raw_root, str) or raw_root == "":
        raise ValueError("root must be a non-empty string path")
    if "\x00" in raw_root:
        raise UnsafePathError("root path must not contain NUL bytes")
    if "\\" in raw_root:
        raise UnsafePathError("root path must not contain backslashes")
    if not raw_root.startswith("/"):
        raise UnsafePathError(f"root path must be absolute, got {raw_root!r}")

    if raw_root == "/":
        return raw_root, []

    parts = raw_root[1:].split("/")
    for part in parts:
        if part == "":
            raise UnsafePathError(
                f"root path {raw_root!r} must not contain empty components or trailing slash"
            )
        if part in (".", ".."):
            raise UnsafePathError(
                f"root path {raw_root!r} must not contain {part!r} components"
            )
    return raw_root, parts


class LocalAdapter:
    """Bounded read-only storage adapter anchored to a POSIX directory descriptor.

    Parameters
    ----------
    source_id:
        Non-empty string identifying this storage source.
    root:
        Canonical absolute path to an existing directory with no symbolic link
        in any path component.
    max_read_bytes:
        Maximum file size in bytes permitted by :meth:`read_bytes` (default
        1,048,576). Must be a positive ``int`` (``bool`` is rejected).
    """

    def __init__(
        self,
        source_id: str,
        root: str | os.PathLike[str],
        *,
        max_read_bytes: int = DEFAULT_MAX_READ_BYTES,
        max_list_entries: int = 10_000,
    ) -> None:
        if not isinstance(source_id, str) or source_id == "":
            raise ValueError("source_id must be a non-empty string")
        if (
            isinstance(max_read_bytes, bool)
            or not isinstance(max_read_bytes, int)
            or max_read_bytes <= 0
        ):
            raise ValueError("max_read_bytes must be a positive integer")
        if (
            isinstance(max_list_entries, bool)
            or not isinstance(max_list_entries, int)
            or max_list_entries < 1
        ):
            raise ValueError("max_list_entries must be a positive integer")

        _check_platform_capabilities()
        raw_root, root_parts = _validate_root_path_string(root)

        self._source_id = source_id
        self._max_read_bytes = max_read_bytes
        self._max_list_entries = max_list_entries
        self._closed = False
        self._root_fd: int | None = None
        self._root_fd = self._open_anchored_root(raw_root, root_parts)

    @property
    def source_id(self) -> str:
        """The source identifier exposed by this adapter."""
        return self._source_id

    @property
    def max_read_bytes(self) -> int:
        """The maximum number of bytes permitted for :meth:`read_bytes`."""
        return self._max_read_bytes

    def close(self) -> None:
        """Close the retained root directory descriptor.

        Idempotent: subsequent calls to :meth:`close` are no-ops. Storage
        operations invoked after :meth:`close` raise :class:`AdapterClosedError`
        (after checking for :class:`SourceMismatchError`).
        """
        if not self._closed:
            self._closed = True
            fd = self._root_fd
            self._root_fd = None
            if fd is not None:
                try:
                    self._os_close(fd)
                except OSError as exc:
                    raise LocalStorageError(
                        f"failed to close root descriptor for {self._source_id!r}: {exc}"
                    ) from exc

    def __enter__(self) -> LocalAdapter:
        if self._closed or self._root_fd is None:
            raise AdapterClosedError(
                f"LocalAdapter for source {self._source_id!r} is closed"
            )
        return self

    def __exit__(
        self,
        exc_type: type[BaseException] | None,
        exc_val: BaseException | None,
        exc_tb: TracebackType | None,
    ) -> None:
        self.close()

    def __del__(self) -> None:
        # Best-effort descriptor cleanup if the caller forgot to close the adapter.
        try:
            self.close()
        except Exception:
            pass

    # -- low-level OS seams (facilitate deterministic unit/adversarial tests) --

    def _os_open(
        self, path: str, flags: int, *, dir_fd: int | None = None
    ) -> int:
        return os.open(path, flags, dir_fd=dir_fd)

    def _os_close(self, fd: int) -> None:
        os.close(fd)

    def _os_fstat(self, fd: int) -> os.stat_result:
        return os.fstat(fd)

    def _os_stat_nofollow(self, path: str, *, dir_fd: int) -> os.stat_result:
        return os.stat(path, dir_fd=dir_fd, follow_symlinks=False)

    def _os_scandir(self, fd: int):
        return os.scandir(fd)

    def _os_read(self, fd: int, length: int) -> bytes:
        return os.read(fd, length)

    # -- descriptor lifetime helpers ------------------------------------------

    def _close_owned_fd(self, fd: int, *, context: str) -> None:
        """Close an owned descriptor on the success path, wrapping OS errors.

        Used where closing the descriptor is part of normal control flow (a
        traversal step succeeded and the previous descriptor is no longer
        needed). Linux releases the descriptor even when ``close`` reports an
        error, so the caller must never retry the close on the same fd.
        """
        try:
            self._os_close(fd)
        except OSError as exc:
            raise LocalStorageError(
                f"failed to close descriptor for {context}: {exc}"
            ) from exc

    def _close_traversal_fd(self, fd: int, *, context: str) -> None:
        """Close a transient descriptor from a ``finally`` block.

        On the normal path a close failure is wrapped in
        :class:`LocalStorageError`. While another exception is already
        propagating the close failure is suppressed, so cleanup can never mask
        the primary storage error (the kernel releases the descriptor
        regardless of the reported error).
        """
        primary = sys.exc_info()[0]
        try:
            self._os_close(fd)
        except OSError as exc:
            if primary is not None:
                return
            raise LocalStorageError(
                f"failed to close descriptor for {context}: {exc}"
            ) from exc

    def _close_quietly(self, fd: int) -> None:
        """Best-effort close used only while unwinding an existing exception."""
        try:
            self._os_close(fd)
        except OSError:
            pass

    def _release_owned(self, owned: list[tuple[int, str]]) -> None:
        """Release each owned descriptor exactly once, in the given order.

        A descriptor is "owned" the moment it is acquired, and is closed exactly
        once here. A failing ``close`` is never retried: Linux releases the
        descriptor even when ``close`` reports an error, and a truly ambiguous
        close cannot be verified from user space, so it must not be retried.
        While another exception is already propagating, close failures are
        suppressed so the primary storage error is never masked; on the success
        path the first close failure is raised as :class:`LocalStorageError`
        with its ``OSError`` cause preserved.
        """
        primary = sys.exc_info()[0]
        first_failure: tuple[str, OSError] | None = None
        for fd, context in owned:
            try:
                self._os_close(fd)
            except OSError as exc:
                if first_failure is None:
                    first_failure = (context, exc)
        if first_failure is not None and primary is None:
            context, exc = first_failure
            raise LocalStorageError(
                f"failed to close descriptor for {context}: {exc}"
            ) from exc

    def _scan_work_error(
        self, entry_count: int, message: str, exc: BaseException
    ) -> StorageError:
        """Build the error for a scan failure, charging already-yielded work.

        Before any raw entry has been yielded the underlying storage error is
        returned unchanged (zero raw work has been done). Once at least one raw
        entry has been yielded the error is wrapped in :class:`ScanFailure`
        carrying the number of successfully yielded raw entries, with the
        original error and its ``OSError`` cause preserved.
        """
        underlying = LocalStorageError(message)
        underlying.__cause__ = exc
        if entry_count > 0:
            return ScanFailure(entry_count, underlying)
        return underlying

    # -- root initialization ---------------------------------------------------

    def _open_anchored_root(self, raw_root: str, parts: list[str]) -> int:
        """Traverse from ``/`` without following symlinks and return root fd.

        Each component is opened atomically with ``O_NOFOLLOW`` (no
        ``O_DIRECTORY``) and the returned descriptor is ``fstat``ed and required
        to be a directory. A component swapped for a symlink immediately before
        its open is rejected by the kernel with ``ELOOP`` (mapped to
        :class:`UnsafePathError`); there is no separate ``lstat`` and therefore
        no check-to-use window.
        """
        dir_flags = _dir_open_flags()
        try:
            current_fd = self._os_open("/", dir_flags)
        except OSError as exc:
            raise self._classify_root_error("/", None, exc, raw_root) from exc

        try:
            try:
                root_st = self._os_fstat(current_fd)
            except OSError as exc:
                raise LocalStorageError(
                    f"failed to fstat '/' while resolving root {raw_root!r}: {exc}"
                ) from exc
            if not stat.S_ISDIR(root_st.st_mode):
                raise ValueError("filesystem '/' is not a directory")

            for part in parts:
                try:
                    next_fd = self._os_open(part, dir_flags, dir_fd=current_fd)
                except OSError as exc:
                    raise self._classify_root_error(
                        part, current_fd, exc, raw_root
                    ) from exc

                try:
                    try:
                        part_st = self._os_fstat(next_fd)
                    except OSError as exc:
                        raise LocalStorageError(
                            f"failed to fstat component {part!r} while "
                            f"resolving root {raw_root!r}: {exc}"
                        ) from exc
                    if not stat.S_ISDIR(part_st.st_mode):
                        raise ValueError(
                            f"root component {part!r} in {raw_root!r} is not a directory"
                        )
                except BaseException:
                    self._close_quietly(next_fd)
                    raise

                # Take ownership of the new descriptor *before* releasing the
                # old one, so a failing close cannot leave the new fd orphaned
                # or cause the old one to be closed twice.
                prev_fd = current_fd
                current_fd = next_fd
                self._close_owned_fd(
                    prev_fd,
                    context=f"root component {part!r} of {raw_root!r}",
                )

            return current_fd
        except BaseException:
            self._close_quietly(current_fd)
            raise

    def _classify_root_error(
        self,
        part: str,
        parent_fd: int | None,
        exc: OSError,
        raw_root: str,
    ) -> Exception:
        """Translate an ``OSError`` during root setup into a contract error."""
        if exc.errno in (errno.ELOOP, getattr(errno, "EMLINK", -1)):
            return UnsafePathError(
                f"symlink rejected at component {part!r} in root {raw_root!r}"
            )
        if parent_fd is not None:
            try:
                st = self._os_stat_nofollow(part, dir_fd=parent_fd)
            except OSError:
                st = None
            if st is not None:
                if stat.S_ISLNK(st.st_mode):
                    return UnsafePathError(
                        f"symlink rejected at component {part!r} in root {raw_root!r}"
                    )
                if not stat.S_ISDIR(st.st_mode):
                    return ValueError(
                        f"root component {part!r} in {raw_root!r} is not a directory"
                    )
        if exc.errno in (errno.ENOENT, errno.ENOTDIR):
            return ValueError(
                f"root {raw_root!r} must be an existing directory: {exc}"
            )
        return LocalStorageError(
            f"failed to open root {raw_root!r} at component {part!r}: {exc}"
        )

    # -- internal helpers ------------------------------------------------------

    def _check_source_and_state(self, ref: FileRef) -> int:
        """Enforce source mismatch first, then adapter open state.

        Returns the active root directory descriptor.
        """
        if not isinstance(ref, FileRef):
            raise TypeError(f"expected FileRef, got {type(ref).__name__}")
        if ref.source_id != self._source_id:
            raise SourceMismatchError(
                f"ref source_id {ref.source_id!r} does not match "
                f"adapter source_id {self._source_id!r}"
            )
        if self._closed or self._root_fd is None:
            raise AdapterClosedError(
                f"LocalAdapter for source {self._source_id!r} is closed"
            )
        return self._root_fd

    def _root_metadata(self) -> FileMetadata:
        return FileMetadata(
            ref=FileRef(self._source_id, ROOT_ID),
            name="",
            parent_id=None,
            kind="directory",
            size_bytes=None,
        )

    def _build_metadata(
        self,
        parts: tuple[str, ...],
        *,
        kind: str,
        size_bytes: int | None,
    ) -> FileMetadata:
        if not parts:
            return self._root_metadata()
        file_id = PATH_PREFIX + "/".join(parts)
        parent_id = (
            ROOT_ID
            if len(parts) == 1
            else PATH_PREFIX + "/".join(parts[:-1])
        )
        return FileMetadata(
            ref=FileRef(self._source_id, file_id),
            name=parts[-1],
            parent_id=parent_id,
            kind=kind,
            size_bytes=size_bytes if kind == "file" else None,
        )

    def _classify_component_open_error(
        self,
        part: str,
        parent_fd: int,
        exc: OSError,
        file_id: str,
        *,
        is_intermediate: bool,
        for_directory_op: bool,
    ) -> Exception:
        """Classify an ``OSError`` from a no-follow ``openat`` on ``part``."""
        if exc.errno == errno.ENOENT:
            return NotFoundError(f"entry {file_id!r} not found")
        if exc.errno in (errno.ELOOP, getattr(errno, "EMLINK", -1)):
            return UnsafePathError(
                f"symlink encountered at {part!r} for {file_id!r}"
            )
        if exc.errno == errno.EISDIR and not for_directory_op and not is_intermediate:
            return IsDirectoryError(f"{file_id!r} is a directory, not a file")

        try:
            st = self._os_stat_nofollow(part, dir_fd=parent_fd)
        except OSError as stat_exc:
            if stat_exc.errno == errno.ENOENT:
                return NotFoundError(f"entry {file_id!r} not found")
            if stat_exc.errno in (errno.ELOOP, getattr(errno, "EMLINK", -1)):
                return UnsafePathError(
                    f"symlink encountered at {part!r} for {file_id!r}"
                )
            st = None

        if st is not None:
            if stat.S_ISLNK(st.st_mode):
                return UnsafePathError(
                    f"symlink encountered at {part!r} for {file_id!r}"
                )
            if stat.S_ISDIR(st.st_mode):
                if not for_directory_op and not is_intermediate:
                    return IsDirectoryError(
                        f"{file_id!r} is a directory, not a file"
                    )
            elif stat.S_ISREG(st.st_mode):
                if is_intermediate or for_directory_op:
                    return NotDirectoryError(
                        f"component {part!r} in {file_id!r} is not a directory"
                    )
            else:
                return UnsupportedEntryError(
                    f"unsupported special filesystem entry at {part!r} for {file_id!r}"
                )

        if exc.errno == errno.ENOTDIR:
            return NotDirectoryError(
                f"component {part!r} in {file_id!r} is not a directory"
            )
        return LocalStorageError(
            f"OS failure accessing {part!r} for {file_id!r}: {exc}"
        )

    def _traverse_parent_dir(
        self, root_fd: int, dir_parts: tuple[str, ...], file_id: str
    ) -> int:
        """Traverse intermediate directory components from ``root_fd``.

        Returns ``root_fd`` when ``dir_parts`` is empty, or a newly opened
        directory descriptor for the final intermediate directory. Each
        component is opened atomically with ``O_NOFOLLOW`` (no ``O_DIRECTORY``)
        and the returned descriptor is ``fstat``ed and required to be a
        directory, so a component swapped for a symlink immediately before its
        open is rejected with ``ELOOP``. Intermediate descriptors opened along
        the way are always closed before returning or raising.
        """
        if not dir_parts:
            return root_fd

        dir_flags = _dir_open_flags()
        current_fd = root_fd
        try:
            for part in dir_parts:
                try:
                    next_fd = self._os_open(part, dir_flags, dir_fd=current_fd)
                except OSError as exc:
                    raise self._classify_component_open_error(
                        part,
                        current_fd,
                        exc,
                        file_id,
                        is_intermediate=True,
                        for_directory_op=True,
                    ) from exc

                try:
                    try:
                        st = self._os_fstat(next_fd)
                    except OSError as exc:
                        raise LocalStorageError(
                            f"failed to fstat component {part!r} for "
                            f"{file_id!r}: {exc}"
                        ) from exc
                    if not stat.S_ISDIR(st.st_mode):
                        if stat.S_ISLNK(st.st_mode):
                            raise UnsafePathError(
                                f"symlink encountered at {part!r} for {file_id!r}"
                            )
                        if stat.S_ISREG(st.st_mode):
                            raise NotDirectoryError(
                                f"component {part!r} in {file_id!r} is not a directory"
                            )
                        raise UnsupportedEntryError(
                            f"unsupported entry at {part!r} for {file_id!r}"
                        )
                except BaseException:
                    self._close_quietly(next_fd)
                    raise

                # Take ownership of the new descriptor before releasing the old
                # one (see _open_anchored_root).
                prev_fd = current_fd
                current_fd = next_fd
                if prev_fd != root_fd:
                    self._close_owned_fd(
                        prev_fd, context=f"component {part!r} of {file_id!r}"
                    )

            return current_fd
        except BaseException:
            if current_fd != root_fd:
                self._close_quietly(current_fd)
            raise

    # -- public StorageAdapter contract ----------------------------------------

    def stat(self, ref: FileRef) -> FileMetadata:
        """Return metadata for ``ref`` without following symbolic links."""
        root_fd = self._check_source_and_state(ref)
        parts = _parse_file_id(ref.file_id)

        if not parts:
            try:
                st = self._os_fstat(root_fd)
            except OSError as exc:
                raise LocalStorageError(
                    f"failed to stat anchored root for {self._source_id!r}: {exc}"
                ) from exc
            if not stat.S_ISDIR(st.st_mode):
                raise LocalStorageError(
                    f"anchored root for {self._source_id!r} is not a directory"
                )
            return self._root_metadata()

        parent_fd = self._traverse_parent_dir(root_fd, parts[:-1], ref.file_id)
        try:
            target_name = parts[-1]
            try:
                st = self._os_stat_nofollow(target_name, dir_fd=parent_fd)
            except OSError as exc:
                if exc.errno == errno.ENOENT:
                    raise NotFoundError(
                        f"entry {ref.file_id!r} not found"
                    ) from exc
                if exc.errno == errno.ENOTDIR:
                    raise NotDirectoryError(
                        f"parent of {ref.file_id!r} is not a directory"
                    ) from exc
                if exc.errno in (errno.ELOOP, getattr(errno, "EMLINK", -1)):
                    raise UnsafePathError(
                        f"symlink encountered for {ref.file_id!r}"
                    ) from exc
                raise LocalStorageError(
                    f"failed to stat {ref.file_id!r}: {exc}"
                ) from exc

            if stat.S_ISLNK(st.st_mode):
                raise UnsafePathError(
                    f"entry {ref.file_id!r} is a symbolic link"
                )
            if stat.S_ISDIR(st.st_mode):

    def scan_children(
        self, parent: FileRef, *, max_entries: int
    ) -> ScanBatch:

        that another entry exists).

    def _read_open_file_bounded(self, file_fd: int, file_id: str) -> bytes:
        """Verify ``file_fd`` is a regular file and read up to ``max_read_bytes``."""
        try:
            st = self._os_fstat(file_fd)
        except OSError as exc:
            raise LocalStorageError(
                f"failed to fstat opened file {file_id!r}: {exc}"
            ) from exc

        if stat.S_ISDIR(st.st_mode):
            raise IsDirectoryError(f"{file_id!r} is a directory, not a file")
        if stat.S_ISLNK(st.st_mode):
            raise UnsafePathError(f"{file_id!r} is a symbolic link")
        if not stat.S_ISREG(st.st_mode):
            raise UnsupportedEntryError(
                f"{file_id!r} is not a regular file (mode {oct(st.st_mode)})"
            )

        if st.st_size > self._max_read_bytes:
            raise ReadLimitExceededError(
                f"file {file_id!r} size ({st.st_size} bytes) exceeds "
                f"max_read_bytes ({self._max_read_bytes} bytes)"
            )

        chunks: list[bytes] = []
        total_read = 0
        max_allowed = self._max_read_bytes
        read_cap = max_allowed + 1

        while True:
            remaining_cap = read_cap - total_read
            if remaining_cap <= 0:
                raise ReadLimitExceededError(
                    f"file {file_id!r} exceeded max_read_bytes "
                    f"({max_allowed} bytes) while reading"
                )
            request_size = min(_READ_CHUNK_SIZE, remaining_cap)
            try:
                chunk = self._os_read(file_fd, request_size)
            except OSError as exc:
                if exc.errno == errno.EINTR:
                    continue
                raise LocalStorageError(
                    f"failed while reading {file_id!r}: {exc}"
                ) from exc

            if not chunk:
                break

            total_read += len(chunk)
            if total_read > max_allowed:
                raise ReadLimitExceededError(
                    f"file {file_id!r} exceeded max_read_bytes "
                    f"({max_allowed} bytes) while reading"
                )
            chunks.append(chunk)

        return b"".join(chunks)

"""Bounded breadth-first name search over the storage contract.

MODULE-005A: :func:`find` walks a storage tree breadth-first, level by level,
and returns matching files and directories while enforcing a raw-entry budget, a
result budget, and a depth bound. It never blocks on a large folder: each
directory is scanned through the adapter's ``scan_children`` with only the
remaining entry budget, so a huge directory is stopped early with
``truncated=True`` rather than fully read.

Search now charges *raw scan work*, not returned metadata. A single bounded
directory scan is charged in full the moment it returns, before any name is
matched, so ``FindResult.inspected`` counts the raw directory entries the
adapter actually examined -- including filtered, unsupported, and vanished
entries -- even when ``max_results`` stops matching early. A partially scanned
folder keeps the matches it did find, but its items are ordered only within the
scanned physical prefix; that is not a claim about the global order of the
directory.

Bounded search requires a callable ``scan_children``: supported-item counts
cannot express raw scan work, so an adapter without it raises
:class:`~drive_tool.storage.ScanUnsupportedError` rather than silently falling
back to ``list_children``.

Python 3.11+, standard library only. No threads, no retries, no sleeps.
"""

from __future__ import annotations

from dataclasses import dataclass

from drive_tool.storage import (
    FileMetadata,
    FileRef,
    ScanFailure,
    ScanUnsupportedError,
    StorageAdapter,
    StorageError,
)

__all__ = ["FindResult", "find", "ScanUnsupportedError"]


@dataclass(frozen=True)
class FindResult:
    """The outcome of a bounded :func:`find`.

    ``items`` are the matches in walk order, ``truncated`` is ``True`` when a
    bound stopped the walk (``reason`` names it), ``skipped`` holds the
    ``file_id`` of every directory that could not be scanned, and ``inspected``
    counts the **raw directory entries examined** by the adapter.

    ``inspected`` is the sum of the ``ScanBatch.inspected`` values charged as
    each directory was scanned, so it reflects the raw scan work even when
    ``max_results`` stopped matching early or when filtered/unsupported/vanished
    entries never produced metadata. It can therefore exceed both the number of
    ``items`` returned and the number of entries actually tested for a match.
    """

    items: tuple[FileMetadata, ...]
    truncated: bool
    reason: str | None
    inspected: int
    skipped: tuple[str, ...]


def _validate_int_in_range(
    name: str, value: object, low: int, high: int
) -> None:
    if isinstance(value, bool) or not isinstance(value, int):
        raise ValueError(
            f"{name} must be an int in {low}..{high}, "
            f"got {type(value).__name__}"
        )
    if value < low or value > high:
        raise ValueError(f"{name} must be in {low}..{high}, got {value}")


def _validate_batch_report(
    batch: object, remaining: int
) -> tuple[tuple[FileMetadata, ...], int, bool]:
    """Validate a scanner's reported batch loudly, without clamping.

    ``items`` must already be a tuple (an arbitrary iterable is never consumed
    to measure it), ``inspected`` a plain ``int`` in ``0..remaining``, and
    ``complete`` a real ``bool``. A batch that claims to be incomplete but did
    not exhaust the remaining budget is rejected rather than silently accepted
    as a no-progress scan.
    """
    items = getattr(batch, "items", None)
    if not isinstance(items, tuple):
        raise StorageError(
            "scan_children must return a ScanBatch whose items is a tuple, "
            f"got {type(items).__name__}"
        )
    inspected = getattr(batch, "inspected", None)
    if isinstance(inspected, bool) or not isinstance(inspected, int):
        raise StorageError(
            f"ScanBatch.inspected must be an int, got {type(inspected).__name__}"
        )
    if inspected < 0 or inspected > remaining:
        raise StorageError(
            f"ScanBatch.inspected {inspected} is outside the range 0..{remaining}"
        )
    complete = getattr(batch, "complete", None)
    if not isinstance(complete, bool):
        raise StorageError(
            f"ScanBatch.complete must be a bool, got {type(complete).__name__}"
        )
    if len(items) > inspected:
        raise StorageError(
            f"ScanBatch reports {len(items)} items but only {inspected} inspected"
        )
    if not complete and inspected != remaining:
        raise StorageError(
            f"an incomplete ScanBatch must exhaust the remaining budget "
            f"{remaining}, but reported inspected={inspected}"
        )
    return items, inspected, complete

def find(
    adapter: StorageAdapter,
    parent: FileRef,
    name_contains: str,
    *,
    max_results: int = 20,
    max_entries: int = 5000,
    max_depth: int = 4,
    case_sensitive: bool = False,
) -> FindResult:
    """Return matches for ``name_contains`` in the subtree rooted at ``parent``.

    The walk is breadth-first; directories within a level are visited in name
    order and matches are returned in walk order (shallower first, then by
    name). A match is a substring test on the entry name (casefolded unless
    ``case_sensitive``), and both files and directories can match.

    All arguments are validated before the adapter is touched. The adapter must
    expose a callable ``scan_children``; otherwise
    :class:`~drive_tool.storage.ScanUnsupportedError` is raised before any
    enumeration is attempted. Each directory is scanned with only the remaining
    entry budget (``max_entries - inspected``), and the batch's ``inspected`` is
    charged immediately, so ``inspected`` measures raw scan work even when
    ``max_results`` stops matching early.

    A ``ScanFailure`` re-raises the original error at the starting root, or, at
    depth ``> 0``, is recorded in ``skipped`` after charging its ``inspected``,
    and the walk continues only while budget remains. Any other ``StorageError``
    listing a subdirectory is recorded in ``skipped`` (charging nothing, since
    it failed before scanning); the same error listing ``parent`` propagates.
    Reaching an entry budget with work left, or scanning a folder that was only
    partially covered, ends the walk with ``truncated=True, reason="max_entries"``
    unless ``max_results`` already stopped it first.

    if not isinstance(name_contains, str) or name_contains == "":
        raise ValueError("name_contains must be a non-empty str")

    _validate_int_in_range("max_entries", max_entries, 1, 100_000)
    _validate_int_in_range("max_depth", max_depth, 0, 16)
    if not isinstance(case_sensitive, bool):
        raise ValueError(

            f"adapter {type(adapter).__name__!r} has no callable scan_children; "
            f"bounded search requires bounded raw scanning"

    current_level: list[tuple[FileRef, int]] = [(parent, 0)]

                inspected += _validate_failure_inspected(failure, remaining_budget)

                    # error object (and its preserved cause).
                    if not isinstance(failure.error, StorageError):
                        raise StorageError(
                            "ScanFailure.error must be a StorageError"

                    raise failure.error

                # continues while budget remains.
                skipped.append(dir_ref.file_id)
                continue
            except StorageError:
                if depth == 0:
                    # The starting parent's own errors propagate unchanged.
                    raise
                # A subdirectory that failed before scanning (or that vanished)
                # is skipped and charges no raw entries.
                skipped.append(dir_ref.file_id)
                continue

                batch, remaining_budget

            more_dirs_at_this_level = (idx + 1) < len(current_level)

                if entry.kind == "directory":
                    if depth < max_depth:

                        hit_max_depth = True

                haystack = (
                    entry.name if case_sensitive else entry.name.casefold()
                )
                if needle in haystack:
                    matches.append(entry)
                    if len(matches) == max_results:
                        more_entries_in_dir = (child_idx + 1) < len(children)
                        more_dirs_queued = (
                            more_dirs_at_this_level or bool(next_level_dirs)
                        )
                        if (
                            more_entries_in_dir
                            or not batch_complete
                            or more_dirs_queued
                        ):
                            return FindResult(
                                items=tuple(matches),
                                truncated=True,
                                reason="max_results",
                                inspected=inspected,
                                skipped=tuple(skipped),
                            )
                        # Exact match count on the last entry of a complete scan
                        # with no queued work: the walk is finished.
                        exact_last_entry = True
                        break

            if exact_last_entry:
                break

            if not batch_complete:
                # A partially scanned folder is truncation, even if it held no
                # matches: incomplete coverage is not "not found".
                return FindResult(
                    items=tuple(matches),
                    truncated=True,
                    reason="max_entries",
                    inspected=inspected,
                    skipped=tuple(skipped),
                )

        if len(matches) == max_results:
            break

        next_level_dirs.sort(key=lambda pair: (pair[0].name, pair[0].ref.file_id))
        current_level = [(meta.ref, d) for meta, d in next_level_dirs]

    if hit_max_depth:
        return FindResult(
            items=tuple(matches),
            truncated=True,
            reason="max_depth",
            inspected=inspected,
            skipped=tuple(skipped),
        )

    return FindResult(
        items=tuple(matches),
        truncated=False,
        reason=None,
        inspected=inspected,
        skipped=tuple(skipped),
    )

class _ScanProbe:
    """Counts and scripts the raw calls a ``scan_children`` makes."""

    def __init__(self, order=None):
        self.order = order
        self.next_attempts = 0
        self.yielded = 0
        self.stat_calls = 0
        self.entered = 0
        self.exited = 0
        # name -> exception raised from DirEntry.stat
        self.stat_error: dict[str, BaseException] = {}
        # raise OSError(EIO) from the (N+1)-th next() attempt once N yielded
        self.iterate_error_after: int | None = None
        # raise OSError(EIO) from the scandir iterator's __exit__
        self.close_error = False


class _ProbeEntry:
    def __init__(self, inner, probe):
        self._inner = inner
        self._probe = probe

    def __getattr__(self, name):
        return getattr(self._inner, name)

    def stat(self, *, follow_symlinks=True):
        self._probe.stat_calls += 1
        exc = self._probe.stat_error.get(self._inner.name)
        if exc is not None:
            raise exc
        return self._inner.stat(follow_symlinks=follow_symlinks)


class _ProbeScandir:
    def __init__(self, inner, probe):
        self._inner = inner
        self._probe = probe
        self._it = None

    def __enter__(self):
        self._probe.entered += 1
        self._inner.__enter__()
        entries = list(self._inner)
        if self._probe.order is not None:
            rank = {name: i for i, name in enumerate(self._probe.order)}
            entries.sort(key=lambda e: rank.get(e.name, len(rank)))
        self._it = iter(entries)
        return self

    def __iter__(self):
        return self

    def __next__(self):
        self._probe.next_attempts += 1
        after = self._probe.iterate_error_after
        if after is not None and self._probe.yielded >= after:
            raise OSError(errno.EIO, "iteration boom")
        entry = next(self._it)
        self._probe.yielded += 1
        return _ProbeEntry(entry, self._probe)

    def __exit__(self, exc_type, exc, tb):
        self._probe.exited += 1
        result = self._inner.__exit__(exc_type, exc, tb)
        if self._probe.close_error:
            raise OSError(errno.EIO, "scandir close boom")
        return result


class OrderedScanAdapter(LocalAdapter):
    """LocalAdapter whose scandir order, stat calls and failures are scripted."""

    def __init__(self, source_id, root, *, order=None, **kwargs):
        self.probe = _ScanProbe(order=order)
        super().__init__(source_id, root, **kwargs)

    def _os_scandir(self, fd):
        return _ProbeScandir(super()._os_scandir(fd), self.probe)

class _ScanProbe:
    def __init__(self, order=None):
        self.order = order
        self.next_attempts = 0
        self.yielded = 0
        self.stat_calls = 0
        self.entered = 0
        self.exited = 0
        self.stat_error: dict[str, BaseException] = {}
        self.iterate_error_after: int | None = None
        self.close_error = False


class _ProbeEntry:
    def __init__(self, inner, probe):
        self._inner = inner
        self._probe = probe

    def __getattr__(self, name):
        return getattr(self._inner, name)

    def stat(self, *, follow_symlinks=True):
        self._probe.stat_calls += 1
        exc = self._probe.stat_error.get(self._inner.name)
        if exc is not None:
            raise exc
        return self._inner.stat(follow_symlinks=follow_symlinks)


class _ProbeScandir:
    def __init__(self, inner, probe):
        self._inner = inner
        self._probe = probe
        self._it = None

    def __enter__(self):
        self._probe.entered += 1
        self._inner.__enter__()
        entries = list(self._inner)
        if self._probe.order is not None:
            rank = {name: i for i, name in enumerate(self._probe.order)}
            entries.sort(key=lambda e: rank.get(e.name, len(rank)))
        self._it = iter(entries)
        return self

    def __iter__(self):
        return self

    def __next__(self):
        self._probe.next_attempts += 1
        after = self._probe.iterate_error_after
        if after is not None and self._probe.yielded >= after:
            raise OSError(errno.EIO, "iteration boom")
        entry = next(self._it)
        self._probe.yielded += 1
        return _ProbeEntry(entry, self._probe)

    def __exit__(self, exc_type, exc, tb):
        self._probe.exited += 1
        result = self._inner.__exit__(exc_type, exc, tb)
        if self._probe.close_error:
            raise OSError(errno.EIO, "scandir close boom")
        return result

    def __init__(self, source_id, root, *, order=None, **kwargs):

        super().__init__(source_id, root, **kwargs)

    def _os_scandir(self, fd):
        return _ProbeScandir(super()._os_scandir(fd), self.probe)

class FakeAdapter:
    """Minimal StorageAdapter for search tests.

    ``children_by_id`` maps a ``file_id`` to its list of ``FileMetadata``.
    ``scan_children`` (the method ``find`` requires) returns a bounded prefix:
    when ``enforce_limit`` is true and a folder holds more than the requested
    ``max_entries`` it returns that many items with ``complete=False``; when
    ``enforce_limit`` is false it returns every item with ``inspected`` set to
    the full length, an over-budget report a caller must reject. ``raise_on``
    maps a ``file_id`` to an exception instance that both listing methods raise
    instead of returning. Every call is recorded in ``calls`` as
    ``(file_id, max_entries)``.
    """

    def __init__(
        self, source_id, children_by_id, *, enforce_limit=True, raise_on=None
    ):
        self.source_id = source_id
        self._children = {k: list(v) for k, v in children_by_id.items()}
        self._enforce_limit = enforce_limit
        self._raise_on = dict(raise_on or {})
        self.calls = []

    def list_children(self, parent, *, max_entries=None):
        self.calls.append((parent.file_id, max_entries))
        if parent.source_id != self.source_id:
            raise SourceMismatchError(
                f"source mismatch: {parent.source_id!r} != {self.source_id!r}"
            )
        if parent.file_id in self._raise_on:
            raise self._raise_on[parent.file_id]
        items = list(self._children.get(parent.file_id, []))
        if (
            self._enforce_limit
            and max_entries is not None
            and len(items) > max_entries
        ):
            raise EnumerationLimitError(max_entries)
        return items

    def scan_children(self, parent, *, max_entries):
        self.calls.append((parent.file_id, max_entries))
        if parent.source_id != self.source_id:
            raise SourceMismatchError(
                f"source mismatch: {parent.source_id!r} != {self.source_id!r}"
            )
        if parent.file_id in self._raise_on:
            raise self._raise_on[parent.file_id]
        items = list(self._children.get(parent.file_id, []))
        if self._enforce_limit and len(items) > max_entries:
            return ScanBatch(
                items=tuple(items[:max_entries]),
                inspected=max_entries,
                complete=False,
            )
        return ScanBatch(items=tuple(items), inspected=len(items), complete=True)

    def stat(self, ref):
        raise NotImplementedError

    def read_bytes(self, ref):
        raise NotImplementedError

## 5. Changed old assertions (and why)

Semantics changed in three places, so three old assertions had to change. No
assertion was deleted, weakened or skipped.

1. **`test_search.py::test_max_results_truncation_vs_exact_last_entry`** —
   `inspected` for the `max_results=2` case changed `2 → 3`. `find` now charges
   the adapter's full bounded scan of the 3-entry root (`batch.inspected == 3`)
   *before* matching, so `inspected` counts raw scan work, not returned matches.
   This is the brief's "result limit still charges full scan" rule.

2. **`test_search.py::test_max_entries_exact_budget_vs_dirs_remaining`** — split
   into three tests because a local folder holding **exactly** the budget is now
   deliberately **incomplete**:
   - `test_max_entries_budget_covering_whole_tree_is_complete`: budget strictly
     greater than the folder size (3 > 2) still observes EOF and is complete
     (`truncated=False`, `inspected=2`).
   - `test_local_exact_budget_is_conservatively_incomplete` (**changed**): a
     2-entry folder with `max_entries=2` now reports `truncated=True,
     reason="max_entries", inspected=2` (was `truncated=False`). Entry `B+1` is
     never probed, so EOF is unknown and "unknown" is not proof of completion.
   - `test_max_entries_exhausted_with_dirs_remaining`: unchanged in substance
     (`truncated=True`, `reason="max_entries"`, `inspected=2`).

3. **`test_search.py::test_large_folder_stops_early_via_adapter_limit`** —
   `inspected` changed `0 → 100`, and the needle changed `"x" → "zzz"`. The old
   `inspected == 0` recorded the MODULE-004 defect (an aborted scan was not
   charged); the bounded scan is now charged in full. The needle change is a
   fixture correction: every `f<i>.txt` contains `"x"` (in `.txt`), so with the
   new "retain partial matches" rule the old needle would stop at
   `max_results`, not at the scan budget; `"zzz"` keeps the test about
   scan-budget truncation, which is what its name asserts.

4. **`test_search.py::test_defensive_guard_against_adapter_ignoring_budget`** —
   renamed `test_over_budget_report_is_rejected_loudly` and now expects a
   `StorageError`. The brief replaces the old silent clamping with loud
   rejection of an over-budget scanner report.

5. **`test_search.py::test_subdirectory_enumeration_limit_stops_not_skipped`** —
   renamed `test_subdirectory_partial_scan_truncates_not_skipped`. The existing
   assertions are preserved; the trigger is now an incomplete batch (partial
   scan) rather than `EnumerationLimitError`, and `inspected == 10` is asserted
